-- Node Postgres "Pega lead da fila" (substitui o "Get row(s) in sheet"). Operation: Execute Query. Sem Query Parameters.
-- Pega 1 lead por execução. FOR UPDATE SKIP LOCKED garante que dois números nunca pegam o mesmo lead.
-- Se um disparo travar no meio, o lead volta para a fila depois de 30 minutos (no máximo 2 tentativas).
UPDATE sdr_fila f
   SET status = 'pegando', pego_em = now(), tentativas = f.tentativas + 1
 WHERE f.id = (
   SELECT id FROM sdr_fila
    WHERE status = 'novo'
       OR (status = 'pegando' AND pego_em < now() - interval '30 minutes' AND tentativas < 2)
    ORDER BY nota DESC, id
    FOR UPDATE SKIP LOCKED
    LIMIT 1)
RETURNING f.id AS fila_id, f.nome_empresa, f.telefone, f.endereco, f.cidade,
          coalesce(f.website, '') AS website, '' AS rating, '' AS reviews,
          f.especialidades, '' AS mensagem, '' AS disparo, f.origem;
